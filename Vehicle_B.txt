var thresholdLeft  = 75
var thresholdRight = 77

# settings
var baseSpeed = 65
var gainNumerator = 10
var gainDenominator = 100
var stopDistance = 3000
var collisionDistance = 400
var dangerDistance = 300     # C detection threshold during overtake
var deadZone = 4

# emergency safety timing
var clearLaneTime = 4500      # time to steer right and leave opposite lane
var waitAfterDanger = 5000    # wait time after clearing C's lane

# global
var leftValue = 0
var rightValue = 0
var error = 0
var correction = 0
var frontSensorsAvg = 0
var phase = 0

# phase:
# 0 = initial/following line
# 1 = pull out left
# 2 = sharp right / align in overtake lane
# 3 = pass
# 4 = merge back
# 5 = straighten
# 6 = emergency give way: return right to clear C lane
# 7 = emergency wait after clearing C lane

# boolean logic
var hasDetectedWall = 0
var carDetected = 0
var dangerDetected = 0
var emergencyStop = 0

# overtake:
# 0 = not complete
# 1 = overtake done
var overtake = 0

# uturn:
# 0 = not started
# 1 = in progress
# 2 = complete
var uturn = 0

# reset
timer.period[0] = 0
timer.period[1] = 0


onevent prox
    if phase == 0 then
        # Normal driving behind A
        callsub checkCar
        callsub followLine

        if carDetected == 1 then
            callsub do_overtake
        end

    elseif phase == 2 then
        # B is entering / aligning in opposite lane
        # Now check if C is coming
        callsub checkDanger

        if dangerDetected == 1 then
            callsub emergencyGiveWay
        end

    elseif phase == 3 then
        # B is passing A in opposite lane
        # If C comes, do not stop in C lane.
        # First clear the lane by moving right.
        callsub checkDanger

        if dangerDetected == 1 then
            callsub emergencyGiveWay
        end
    end


sub do_overtake
    # Phase 1: pull out left
    callsub leftBlinker
    motor.left.target = baseSpeed
    motor.right.target = 2 * baseSpeed
    timer.period[0] = 7000
    phase = 1


onevent timer0
    if phase == 1 then
        # Phase 2: sharp right / align in overtake lane
        callsub rightBlinker
        motor.left.target = 2 * baseSpeed
        motor.right.target = baseSpeed
        timer.period[0] = 7000
        phase = 2

    elseif phase == 2 then
        # Phase 3: pass A in overtake lane
        callsub clearLeds
        motor.left.target = 2 * baseSpeed
        motor.right.target = 2 * baseSpeed
        timer.period[0] = 30000
        phase = 3

    elseif phase == 3 then
        # Phase 4: merge back to original/right lane
        callsub rightBlinker
        motor.left.target = 2 * baseSpeed
        motor.right.target = baseSpeed
        timer.period[0] = 7000
        phase = 4

    elseif phase == 4 then
        # Phase 5: straighten in original/right lane
        callsub leftBlinker
        motor.left.target = baseSpeed
        motor.right.target = 2 * baseSpeed
        timer.period[0] = 6500
        phase = 5

    elseif phase == 5 then
        # Overtake complete: continue normal line following
        callsub allLightsOn
        motor.left.target = baseSpeed
        motor.right.target = baseSpeed
        timer.period[0] = 0
        overtake = 1
        phase = 0


    elseif phase == 6 then
        # phase 4: straighten yourself in overtake lane
        callsub leftBlinker
        motor.left.target = baseSpeed
        motor.right.target = 2 * baseSpeed
        phase = 7
        timer.period[0] = 6500
        
    elseif phase == 7 then
        # Phase 5: pass: (go straight in original lane)
        callsub allLightsOn
        motor.left.target = baseSpeed
        motor.right.target = baseSpeed
        timer.period[0] = 0
        timer.period[1] = 10000
    	phase =8
    elseif phase == 8 then
        # Emergency finished:
        # Resume normal driving. If A is still ahead,
        # B can attempt overtake again later.
        callsub clearLeds
        timer.period[0] = 0
        dangerDetected = 0
        emergencyStop = 0
        carDetected = 0
        overtake = 0
        phase = 0
        callsub followLine
    end



sub emergencyGiveWay
    # Important:
    # Do NOT stop in the opposite lane.
    # First steer right to leave C's lane, then stop.
    emergencyStop = 1
    callsub rightBlinker

    # Right turn: left motor faster, right motor slower
    motor.left.target = 2 * baseSpeed
    motor.right.target = baseSpeed
 	phase = 6
    timer.period[0] = 7000
   


sub checkDanger
    # Detect vehicle C coming from the opposite lane.
    # We use left/front/center sensors only.
    # This avoids confusing vehicle A, which is usually on B's right side.
    dangerDetected = 0

    if prox.horizontal[0] > dangerDistance then
        dangerDetected = 1
    end

    if prox.horizontal[1] > dangerDistance then
        dangerDetected = 1
    end

    if prox.horizontal[2] > dangerDistance then
        dangerDetected = 1
    end


sub followLine
    callsub checkWall

    if hasDetectedWall == 0 then
        callsub clearLeds

        # Read the current ground sensor values
        leftValue = prox.ground.delta[0]
        rightValue = prox.ground.delta[1]

        # Calculate the line-following error
        error = (leftValue - thresholdLeft) - (rightValue - thresholdRight)

        # Proportional correction
        correction = (error * gainNumerator) / gainDenominator

        # Dead zone for smooth movement
        if correction > -deadZone and correction < deadZone then
            correction = 0
        end

        # Adjust motor speeds
        motor.left.target = baseSpeed - correction
        motor.right.target = baseSpeed + correction
    else
        # Wall detected
        callsub brakeBlinker
        motor.left.target = 0
        motor.right.target = 0
    end


sub checkCar
    # Detect vehicle A in front during normal following
    if prox.horizontal[2] > collisionDistance then
        carDetected = 1
    else
        carDetected = 0
    end


sub checkWall
    frontSensorsAvg = (prox.horizontal[0] + prox.horizontal[1] + prox.horizontal[2] + prox.horizontal[3] + prox.horizontal[4]) / 5

    if frontSensorsAvg > stopDistance then
        hasDetectedWall = 1
    else
        hasDetectedWall = 0
    end


onevent buttons
    if button.left == 1 then
        callsub leftBlinker
        motor.left.target = 0
        motor.right.target = 100

    elseif button.right == 1 then
        callsub rightBlinker
        motor.left.target = 100
        motor.right.target = 0

    elseif button.center == 1 then
        callsub clearLeds
        motor.left.target = 0
        motor.right.target = 0
    end


sub clearLeds
    call leds.top(0, 0, 0)
    call leds.circle(0, 0, 0, 0, 0, 0, 0, 0)


sub leftBlinker
    callsub clearLeds
    call leds.circle(0, 0, 0, 0, 0, 0, 32, 0)


sub rightBlinker
    callsub clearLeds
    call leds.circle(0, 0, 32, 0, 0, 0, 0, 0)


sub brakeBlinker
    callsub clearLeds
    call leds.top(32, 0, 0)


sub allLightsOn
    call leds.top(32, 32, 32)
    call leds.circle(32, 32, 32, 32, 32, 32, 32, 32)
    call sound.freq(440, 10000)
